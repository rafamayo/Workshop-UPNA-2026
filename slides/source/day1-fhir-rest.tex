	\section{FHIR Fundamentals}
	
	\begin{frame}{What is FHIR?}
		\begin{block}{FHIR = Fast Healthcare Interoperability Resources}
			\begin{itemize}
				\item Modern standard by HL7
				\item Resource-based, composable building blocks
				\item Designed for web technologies (HTTP, REST, JSON/XML)
				\item Supports many use cases:
				\begin{itemize}
					\item Patient administration
					\item Orders \& results
					\item Observations \& vital signs
					\item Care plans, medication, etc.
				\end{itemize}
			\end{itemize}
		\end{block}
		\vspace{0.5em}
		\small Reference: \url{https://www.hl7.org/fhir/}
	\end{frame}
	
	%=====================================================
	\begin{frame}{Resources: The Basic Building Blocks}
		\begin{itemize}
			\item A \textbf{Resource} = small, well-defined data structure
			\item Examples:
			\begin{itemize}
				\item \texttt{Patient}
				\item \texttt{Observation}
				\item \texttt{Condition}
				\item \texttt{DiagnosticReport}
			\end{itemize}
			\item Each resource has:
			\begin{itemize}
				\item \texttt{resourceType}
				\item \texttt{id}
				\item metadata (\texttt{meta}, \texttt{text}, ...)
				\item domain-specific data
			\end{itemize}
		\end{itemize}
		% TODO: add a simple UML-like diagram showing Patient, Observation, DiagnosticReport
	\end{frame}
	
	%=====================================================
	\begin{frame}[fragile]{Example: Patient (JSON)}
		\lstset{language=JSON}
		\begin{lstlisting}
{
	"resourceType": "Patient",
	"id": "12345",
	"name": [{
		"family": "Doe",
		"given": ["John"]
	}],
	"gender": "male",
	"birthDate": "1990-01-01"
}
		\end{lstlisting}
		\begin{itemize}
			\item Minimal example
			\item Real-world patients have more attributes (identifiers, address, ...)
			\item \url{https://www.hl7.org/fhir/patient.html}
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}[fragile]{Example: Observation (JSON)}
		\lstset{language=JSON}
		\begin{lstlisting}
{
	"resourceType": "Observation",
	"status": "final",
	"code": {
		"coding": [{
			"system": "http://loinc.org",
			"code": "85354-9",
			"display": "Blood pressure panel"
		}]
	},
	"subject": { "reference": "Patient/12345" },
	"effectiveDateTime": "2024-05-10T10:00:00Z",
	"valueQuantity": {
		"value": 120,
		"unit": "mmHg"
	}
}
		\end{lstlisting}
		\begin{itemize}
			\item Reference patient via \texttt{subject.reference}; Use LOINC for coding the measurement type
			\item \url{https://www.hl7.org/fhir/observation.html}
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}{References and Linking}
		\begin{itemize}
			\item Resources can reference each other:
			\begin{itemize}
				\item \texttt{Observation.subject} $\rightarrow$ \texttt{Patient}
				\item \texttt{DiagnosticReport.result} $\rightarrow$ \texttt{Observation}
			\end{itemize}
			\item Enables modeling of clinical workflows:
			\begin{itemize}
				\item Admission, diagnostic tests, reports, follow-up
			\end{itemize}
		\end{itemize}

	\section{RESTful APIs for FHIR}
	
	\begin{frame}{REST in a Nutshell}
		\begin{itemize}
			\item REST = Representational State Transfer
			\item Uses HTTP verbs:
			\begin{description}
				\item[GET] read / search
				\item[POST] create
				\item[PUT] replace / update
				\item[DELETE] delete
			\end{description}
			\item Resources are addressed by URLs:
			\begin{itemize}
				\item \texttt{/Patient/123}
				\item \texttt{/Observation?code=XYZ\&subject=Patient/123}
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}[fragile]{Simple HTTP Examples}
		\lstset{language=HTTP}
		\begin{lstlisting}
GET /fhir/Patient/123 HTTP/1.1
Host: example.org
Accept: application/fhir+json
		\end{lstlisting}
		
		\begin{lstlisting}
POST /fhir/Patient HTTP/1.1
Host: example.org
Content-Type: application/fhir+json
			
{ ...patient JSON body... }
		\end{lstlisting}
		\begin{itemize}
			\item FHIR defines how servers must behave when these requests are sent
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}{FHIR RESTful Operations}
		\begin{itemize}
			\item Instance-level:
			\begin{itemize}
				\item \texttt{GET /Patient/123}
				\item \texttt{PUT /Patient/123}
				\item \texttt{DELETE /Patient/123}
			\end{itemize}
			\item Type-level:
			\begin{itemize}
				\item \texttt{POST /Patient}
				\item \texttt{GET /Patient?name=smith}
			\end{itemize}
			\item Whole-system:
			\begin{itemize}
				\item \texttt{GET /metadata}
				\item \texttt{GET /\_history}
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}{Search Parameters}
		\begin{itemize}
			\item FHIR defines a set of search parameters per resource type
			\item Examples:
			\begin{itemize}
				\item \texttt{Patient?name=smith}
				\item \texttt{Observation?subject=Patient/123}
				\item \texttt{Observation?code=85354-9}
				\item \texttt{Observation?date=gt2024-01-01}
			\end{itemize}
			\item Can be combined:
			\begin{itemize}
				\item \texttt{Observation?code=85354-9\&subject=Patient/123}
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	%=====================================================
	\begin{frame}{Error Handling}
		\begin{itemize}
			\item HTTP status codes:
			\begin{itemize}
				\item 200 OK
				\item 201 Created
				\item 400 Bad Request
				\item 404 Not Found
				\item 422 Unprocessable Entity
			\end{itemize}
			\item FHIR may return an \texttt{OperationOutcome} resource describing the problem
			\item In the lab we will intentionally break some JSON to see how the server responds
		\end{itemize}
	\end{frame}
